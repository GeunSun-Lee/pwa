📦 root (GitHub Pages 배포 폴더)
├── index.html          ← 이 파일
├── style.css           ← 전체 스타일 (CSS Variables, Layout, Components)
├── app.js              ← 메인 진입점 (라우터, 스토어, 컴포넌트 등록)
├── db.js               ← IndexedDB 래퍼 (ReadingDB 클래스/함수 export)
├── views/
│   ├── ListView.js     ← 메인 리스트 (그리드/리스트, 무한스크롤, 필터)
│   ├── FormView.js     ← 등록/수정 폼 (ISBN 조회, 이미지 업로드)
│   ├── DetailView.js   ← 상세/리뷰/메모
│   ├── StatsView.js    ← 통계 대시보드 (Canvas 차트)
│   └── SettingsView.js ← 설정/백업/복원/테마
├── components/
│   ├── BookCard.js     ← 도서 카드 컴포넌트
│   ├── BookForm.js     ← 폼 컴포넌트
│   ├── Header.js       ← 상단 헤더 (검색, 뷰 토글, 테마)
│   ├── Modal.js        ← 모달 래퍼
│   └── Toast.js        ← 토스트 알림
├── utils/
│   ├── router.js       ← 해시 라우터 (parseHash, navigate)
│   ├── store.js        ← 전역 상태 (Proxy 반응형)
│   ├── date.js         ← 날짜 포맷팅
│   └── export.js       ← JSON/CSV/이미지 내보내기 헬퍼
├── manifest.json       ← PWA 매니페스트
├── sw.js               ← Service Worker (Workbox 미사용, 순수 SW)
└── icons/              ← PWA 아이콘 세트 (192, 512, maskable)
